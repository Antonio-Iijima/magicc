#require datatypes.numeric
#require datatypes.string

#define single_comment ";[^\n]*\n+"
#define multi_comment "#\|[\s\S]*?\|#"


PROGRAM   -> LISP_EXPR | LISP_EXPR PROGRAM
LISP_EXPR -> "'" S_EXPR | S_EXPR
S_EXPR    -> ATOM | LIST

ATOM      -> DATATYPES.NUMERIC.NUMBER | DATATYPES.STRING.STRING | OPERATOR | "[_a-zA-Z][^\s\'()]*"

LIST      -> "\(" "\)" | "\(" ELEMS "\)"
ELEMS     -> LISP_EXPR | LISP_EXPR ELEMS

OPERATOR  -> "[+*/<>%?]" | "\-" | "//" | "\*\*" | "<=" | ">=" | "=="
