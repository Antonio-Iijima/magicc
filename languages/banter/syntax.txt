#require datatypes.identifier
#require datatypes.numeric
#require datatypes.string
#require math.infix
#require stdlib


PROGRAM          -> INSTRUCTIONS
INSTRUCTIONS     -> INSTRUCTIONS STMT | STMT | EXPRESSION
STMT             -> STATEMENT STDLIB.NEWLINE | STATEMENT
STMTS            -> STMTS STMT | STMT

STATEMENT        -> ASSIGNMENT | CONDITIONAL | OUTPUT | GOTO | MARKER | PRINT | STDLIB.COMMENT

ASSIGNMENT       -> "let" DATATYPES.IDENTIFIER.ID "be" EXPRESSION

CONDITIONAL      -> IF_THEN | IF_THEN_ELSE
IF_THEN          -> "if" EXPRESSION "," "then" BLOCK
IF_THEN_ELSE     -> "if" EXPRESSION "," "then" BLOCK "else" BLOCK
BLOCK            -> STDLIB.NEWLINE INDENT STMTS DEDENT | STMT

OUTPUT           -> "output" EXPRESSION

GOTO             -> "goto" "instruction" MARK
MARKER           -> "@" MARK
MARK             -> DATATYPES.IDENTIFIER.ID | DATATYPES.NUMERIC.NUMBER

PRINT            -> "print" EXPRESSION | "print"

EXPRESSION       -> MATH.INFIX.INFIX
MATH.INFIX.VALUE -> DATATYPES.STRING.STRING | LABEL
LABEL            -> DATATYPES.IDENTIFIER.ID
